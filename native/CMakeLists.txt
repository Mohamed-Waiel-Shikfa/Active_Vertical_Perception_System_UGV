cmake_minimum_required(VERSION 3.16)
project(greenhouse_native LANGUAGES CXX)
cmake_policy(SET CMP0100 NEW)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(gz_gui_vendor REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-gui8 REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Quick Qml QuickControls2 Test)
add_library(GreenhouseSystems SHARED Systems.cc GimbalMath.hh)
target_link_libraries(GreenhouseSystems PRIVATE gz-sim8::gz-sim8 Qt5::Core)
add_library(JackalControls SHARED JackalControls.cc JackalControls.hh GuiChecks.cc OperatorChecks.cc controls.qrc)
target_link_libraries(JackalControls PRIVATE gz-gui8::gz-gui8 Qt5::Quick Qt5::Qml Qt5::QuickControls2 Qt5::Test)
target_link_options(JackalControls PRIVATE "-Wl,--no-undefined")
install(TARGETS GreenhouseSystems JackalControls LIBRARY DESTINATION lib)
add_executable(greenhouse-service ServiceClient.cc)
target_link_libraries(greenhouse-service PRIVATE gz-sim8::gz-sim8 Qt5::Core)
install(TARGETS greenhouse-service RUNTIME DESTINATION bin)
add_executable(qml-contract QmlContract.cc)
target_link_libraries(qml-contract PRIVATE JackalControls gz-gui8::gz-gui8 Qt5::Quick Qt5::Qml Qt5::Test)
install(TARGETS qml-contract RUNTIME DESTINATION bin)
add_executable(gimbal-math-test GimbalMathTest.cc)
